82template<typename StoredValue_, typename StoredIndex_, typename InputValue_, typename InputIndex_>
83FragmentedSparseContents<StoredValue_, StoredIndex_> retrieve_fragmented_sparse_contents_consistent(
84 const Matrix<InputValue_, InputIndex_>& matrix,
85 const bool row,
86 const RetrieveFragmentedSparseContentsOptions& options
87) {
88 const InputIndex_ NR = matrix.nrow();
89 const InputIndex_ NC = matrix.ncol();
90 const InputIndex_ primary = (row ? NR : NC);
91 const InputIndex_ secondary = (row ? NC : NR);
92
93 FragmentedSparseContents<StoredValue_, StoredIndex_> output(primary);
94 auto& store_v = output.value;
95 auto& store_i = output.index;
96
97 if (matrix.is_sparse()) {
98 parallelize([&](const int, const InputIndex_ start, const InputIndex_ length) -> void {
99 auto wrk = consecutive_extractor<true>(matrix, row, start, length);
100 auto buffer_v = create_container_of_Index_size<std::vector<InputValue_> >(secondary);
101 auto buffer_i = create_container_of_Index_size<std::vector<InputIndex_> >(secondary);
102
103 for (InputIndex_ p = start, pe = start + length; p < pe; ++p) {
104 const auto range = wrk->fetch(buffer_v.data(), buffer_i.data());
105 auto& sv = store_v[p];
106 auto& si = store_i[p];
107 sv.reserve(range.number);
108 si.reserve(range.number);
109
110 // We don't filter out structural non-zeros that have values of zero, for consistency with convert_to_compressed_sparse().
111 for (InputIndex_ i = 0; i < range.number; ++i) {
112 sv.push_back(range.value[i]);
113 si.push_back(range.index[i]);
114 }
115 }
116 }, primary, options.num_threads);
117
118 } else {
119 parallelize([&](const int, const InputIndex_ start, const InputIndex_ length) -> void {
120 auto wrk = consecutive_extractor<false>(matrix, row, start, length);
121 auto buffer_v = create_container_of_Index_size<std::vector<InputValue_> >(secondary);
122
123 for (InputIndex_ p = start, pe = start + length; p < pe; ++p) {
124 const auto ptr = wrk->fetch(buffer_v.data());
125 auto& sv = store_v[p];
126 auto& si = store_i[p];
127
128 // For dense, we do treat zero values as structural zeros and remove them, otherwise the output wouldn't actually be sparse.
129 for (InputIndex_ s = 0; s < secondary; ++s) {
130 const auto val = ptr[s];
131
132 if (val) {
133 sv.push_back(val);
134 si.push_back(s);
135 }
136 }
137 }
138 }, primary, options.num_threads);
139 }
140
141 return output;
142}

144template<typename StoredValue_, typename StoredIndex_, typename InputValue_, typename InputIndex_>
145FragmentedSparseContents<StoredValue_, StoredIndex_> retrieve_fragmented_sparse_inconsistent_one_pass(
146 const tatami::Matrix<InputValue_, InputIndex_>& matrix,
147 const bool row,
148 const int num_threads
149) {
150 const InputIndex_ NR = matrix.nrow();
151 const InputIndex_ NC = matrix.ncol();
152 const InputIndex_ primary = (row ? NR : NC);
153 const InputIndex_ secondary = (row ? NC : NR);
154
155 // In the one-pass strategy, we load everything in a nice format first, then we transpose it in serial.
156 // This avoids messy reallocations when trying to expand vectors on an inconsistent dimension.
157 auto store_v = create_container_of_Index_size<std::vector<std::vector<InputValue_> > >(secondary);
158 auto store_i = create_container_of_Index_size<std::vector<std::vector<InputIndex_> > >(secondary);
159 auto original_ranges = extract_sparse_non_zeros_inconsistent(
160 matrix,
161 primary,
162 secondary,
163 row,
164 store_v,
165 store_i,
166 num_threads
167 );
168
169 auto primary_counts = create_container_of_Index_size<std::vector<InputIndex_> >(primary);
170 for (I<decltype(secondary)> s = 0; s < secondary; ++s) {
171 const auto& sec_indices = original_ranges[s].index;
172 const auto num = original_ranges[s].number;
173 for (I<decltype(num)> n = 0; n < num; ++n) {
174 primary_counts[sec_indices[n]] += 1; // addition must be safe as this cannot exceed dimension extents.
175 }
176 }
177
178 FragmentedSparseContents<StoredValue_, StoredIndex_> output(primary);
179 for (InputIndex_ p = 0; p < primary; ++p) {
180 output.index[p].reserve(primary_counts[p]);
181 output.value[p].reserve(primary_counts[p]);
182 }
183
184 for (I<decltype(secondary)> s = 0; s < secondary; ++s) {
185 const auto& sec_values = original_ranges[s].value;
186 const auto& sec_indices = original_ranges[s].index;
187 const auto num = original_ranges[s].number;
188 for (I<decltype(num)> n = 0; n < num; ++n) {
189 const auto curp = sec_indices[n];
190 output.value[curp].push_back(sec_values[n]);
191 output.index[curp].push_back(s);
192 }
193 }
194
195 return output;
196}

198template<typename StoredValue_, typename StoredIndex_, typename InputValue_, typename InputIndex_>
199FragmentedSparseContents<StoredValue_, StoredIndex_> retrieve_fragmented_sparse_inconsistent_two_pass(
200 const tatami::Matrix<InputValue_, InputIndex_>& matrix,
201 const bool row,
202 const int num_threads
203) {
204 const InputIndex_ NR = matrix.nrow();
205 const InputIndex_ NC = matrix.ncol();
206 const InputIndex_ primary = (row ? NR : NC);
207 const InputIndex_ secondary = (row ? NC : NR);
208
209 // In the two-pass strategy, we count the number of non-zeros first, then we fill it up in the second pass.
210 std::optional<std::vector<InputIndex_> > nnz_consistent;
211 auto nnz_inconsistent = create_container_of_Index_size<std::vector<InputIndex_> >(primary);
212 count_sparse_non_zeros_inconsistent(matrix, primary, secondary, row, nnz_inconsistent.data(), nnz_consistent, num_threads);
213
214 FragmentedSparseContents<StoredValue_, StoredIndex_> output(primary);
215 for (InputIndex_ p = 0; p < primary; ++p) {
216 output.index[p].reserve(nnz_inconsistent[p]);
217 output.value[p].reserve(nnz_inconsistent[p]);
218 }
219
220 fill_sparse_matrix_inconsistent(
221 matrix,
222 primary,
223 secondary,
224 row,
225 nnz_inconsistent,
226 /* sparse_main = */ [&](const InputIndex_ s, const SparseRange<InputValue_, InputIndex_>& range) -> void {
227 for (InputIndex_ i = 0; i < range.number; ++i) {
228 output.value[range.index[i]].push_back(range.value[i]);
229 output.index[range.index[i]].push_back(s);
230 }
231 },
232 /* dense_main = */ [&](const InputIndex_ s, const InputValue_* const ptr) -> void {
233 for (InputIndex_ p = 0; p < primary; ++p) {
234 const auto val = ptr[p];
235 if (val != 0) {
236 output.value[p].push_back(val);
237 output.index[p].push_back(s);
238 }
239 }
240 },
241 /* reduce = */ [&](const InputIndex_ s, const std::vector<InputValue_>& cur_values, const std::vector<InputIndex_>& cur_primary_indices) {
242 const auto cur_count = cur_values.size();
243 for (I<decltype(cur_count)> i = 0; i < cur_count; ++i) {
244 const auto primary = cur_primary_indices[i];
245 output.value[primary].push_back(cur_values[i]);
246 output.index[primary].push_back(s);
247 }
248 },
249 num_threads
250 );
251
252 return output;
253}
270template<typename StoredValue_, typename StoredIndex_, typename InputValue_, typename InputIndex_>
271FragmentedSparseContents<StoredValue_, StoredIndex_> retrieve_fragmented_sparse_contents(
272 const Matrix<InputValue_, InputIndex_>& matrix,
273 const bool row,
274 const RetrieveFragmentedSparseContentsOptions& options
275) {
276 if (row == matrix.prefer_rows()) {
