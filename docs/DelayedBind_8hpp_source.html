36/*
37 * Consider the problem of mapping an index 'i' of the combined dimension of the DelayedBind object to the component matrix.
38 * The simplest approach is to do a binary search on 'cumulative' but that has O(log(n)) scaling for 'n' component matrices.
39 * Another approach is to create a look-up table that can be indexed directly by 'i', which has O(1) runtime but consumes a lot of memory.
40 *
41 * The CompressedMapping class is a compromise that breaks up the combined dimensions into bins of regular size no greater than the extent of the smallest component matrix.
42 * Each bin stores the identity the component matrix in which it starts, plus the length of that component matrix covered by that bin.
43 * Given 'i', we can divide by the bin size to find its matching bin; we then compare the remainder to the covered length to assign it to the starting or next matrix.
44 * This has O(1) runtime while being much more memory efficient for a minimum bin size >> 1.
45 *
46 * In practice, the bin size is set to the largest power of 2 that is no greater than the small matrix extent.
47 * This allows for (i) some very fast division via bit shifting, and (ii) bitpacking to store both the matrix identity and the covered length in a single integer.
48 */
49template<typename Index_>
50class CompressedMapping {
51public:
52 CompressedMapping() = default;
53
54 CompressedMapping(Index_ min_size, const std::vector<Index_>& cumulative) {
55 // Define bin size as the smallest power of 2 that is no greater than 'min_size'.
56 // This means that 'my_shift' must be less than the number of binary digits of 'Index_'.
57 while (min_size > 1) {
58 ++my_shift;
59 min_size /= 2;
60 }
61 assert(sanisizer::is_less_than(min_size, std::numeric_limits<Index_>::digits));
62 const auto bin_size = static_cast<Index_>(1) << my_shift;
63 my_mask = (bin_size - 1);
64
65 const auto total_dim = cumulative.back();
66 assert(min_size <= total_dim);
67 const auto total_bins = (total_dim >> my_shift) + ((total_dim & my_mask) > 0);
68 my_bins.reserve(total_bins);
69
70 Index_ pos = 1;
71 for (I<decltype(total_bins)> b = 0; b < total_bins; ++b) {
72 // Find the first cumulative boundary that exceeds the start of the bin.
73 // 'pos - 1' is the index of the matrix covering the start of the bin.
74 Index_ lower = b << my_shift;
75 while (cumulative[pos] <= lower) {
76 ++pos;
77 }
78
79 // We compute the gap between the start of the bin and its right-side boundary.
80 // This is the number of elements in the bin that are assigned to the 'pos - 1'-th matrix.
81 const auto until_next = sanisizer::min(cumulative[pos] - lower, bin_size);
82
83 // Now we pack both of these values into a single integer to save some space.
84 // This is possible based on the following logic:
85 //
86 // - The total cumulative dimension is less than '2^X' where 'X' is the number of digits of 'Index_'.
87 // - The minimum size of each matrix is greater than or equal to '2^Y', where 'Y' is 'my_shift'.
88 // We note that 'X > Y' as the minimum size cannot be greater than the total cumulative dimension.
89 // - The maximum possible value of the number of matrices must be less than '2^(X - Y)'.
90 // This implies that the matrix index must fit into a '(X - Y)'-bit unsigned integer.
91 // - We also know that 'until_next' must be less than or equal to '2^Y'.
92 // This implies that 'until_next - 1' must fit into a Y-bit unsigned integer.
93 //
94 // Thus, we can shift the matrix index by 'Y' and use the cleared bits to store 'until_next - 1'.
95 // We effectively use 'Index_' to pack the two unsigned integers.
96 const Index_ encoding = (static_cast<Index_>(pos - 1) << my_shift) | static_cast<Index_>(until_next - 1);
97 my_bins.push_back(encoding);
98 }
99 }
100
101 Index_ operator[](const Index_ i) const {
102 const auto encoded = my_bins[i >> my_shift];
103 // 'encoded >> my_shift' recovers 'pos - 1'.
104 // 'encoded & my_mask' recovers 'until_next - 1'.
105 // We want to know if 'i & my_mask >= until_next', which is the same as 'i & my_mask > until_next - 1'.
106 return sanisizer::sum_unsafe<Index_>(encoded >> my_shift, sanisizer::is_greater_than(i & my_mask, encoded & my_mask));
107 }
108
109public:
110 std::vector<Index_> my_bins;
111 Index_ my_shift = 0;
112 Index_ my_mask;
113};

119template<typename Index_, class Initialize_>
120Index_ initialize_parallel_block(
121 const std::vector<Index_>& cumulative,
122 const CompressedMapping<Index_>& mapping,
123 const Index_ block_start,
124 const Index_ block_length,
125 const Initialize_ init
126) {
127 const Index_ nmats = cumulative.size() - 1; // Number of matrices is guaranteed to fit in Index_, see reasoning in the DelayedBind constructor.
128 if (nmats == 0) {
129 return 0;
130 }
131
132 const Index_ start_index = mapping[block_start];
133 Index_ actual_start = block_start - cumulative[start_index];
134 const Index_ block_end = block_start + block_length;
135
136 for (Index_ index = start_index; index < nmats; ++index) {
137 const Index_ submat_end = cumulative[index + 1];
138 bool not_final = (block_end > submat_end);
139 const Index_ actual_end = (not_final ? submat_end : block_end) - cumulative[index];
140 init(index, actual_start, actual_end - actual_start);
141 if (!not_final) {
142 break;
143 }
144 actual_start = 0;
145 }
146
147 return start_index;
148}

150template<typename Index_, class Initialize_>
151void initialize_parallel_index(
152 const std::vector<Index_>& cumulative,
153 const CompressedMapping<Index_>& mapping,
154 const std::vector<Index_>& indices,
155 const Initialize_ init
156) {
157 Index_ counter = 0;
158 const Index_ il = indices.size();
159 while (counter < il) {
160 const Index_ first_index = indices[counter];
161 const Index_ bind_index = mapping[first_index];
162 const Index_ lower = cumulative[bind_index];
163 const Index_ upper = cumulative[bind_index + 1];
164
165 // Creating the slice with one element already.
166 auto slice_ptr = std::make_shared<std::vector<Index_> >(1, first_index - lower);
167 ++counter;
168
169 while (counter < il && indices[counter] < upper) {
170 slice_ptr->push_back(indices[counter] - lower);
171 ++counter;
172 }
173
174 init(bind_index, std::move(slice_ptr));
175 }
176}

506template<typename Index_, class Initialize_>
507void initialize_perp_oracular(
508 const std::vector<Index_>& cumulative,
509 const CompressedMapping<Index_>& mapping,
510 const Oracle<Index_>& oracle,
511 std::vector<Index_>& chosen,
512 Initialize_ init
513) {
514 const auto ntotal = oracle.total();
515 chosen.reserve(ntotal);
516
517 struct Predictions {
518 bool consecutive = true;
519 Index_ start = 0;
520 Index_ number = 0;
521 std::vector<Index_> predictions;
522
523 void add(const Index_ p) {
524 if (consecutive) {
525 if (number == 0) {
526 start = p;
527 number = 1;
528 return;
529 }
530 if (number + start == p) {
531 ++number;
532 return;
533 }
534 consecutive = false;
535 resize_container_to_Index_size(predictions, number);
536 std::iota(predictions.begin(), predictions.end(), start);
537 }
538
539 predictions.push_back(p);
540 }
541 };
542
543 const auto nmats = cumulative.size() - 1;
544 auto predictions = create_container_of_Index_size<std::vector<Predictions> >(nmats); // nmats should fit in an Index_, so this call is legal.
545 for (I<decltype(ntotal)> i = 0; i < ntotal; ++i) {
546 const auto prediction = oracle.get(i);
547 const Index_ choice = mapping[prediction];
548 chosen.push_back(choice);
549 predictions[choice].add(prediction - cumulative[choice]);
550 }
551
552 for (I<decltype(nmats)> x = 0; x < nmats; ++x) {
553 auto& current = predictions[x];
554 if (current.consecutive) {
555 if (current.number) {
556 init(x, std::make_shared<ConsecutiveOracle<Index_> >(current.start, current.number));
557 }
558 } else {
559 if (!current.predictions.empty()) {
560 init(x, std::make_shared<FixedVectorOracle<Index_> >(std::move(current.predictions)));
561 }
562 }
563 }

654template<typename Value_, typename Index_>
655class DelayedBind final : public Matrix<Value_, Index_> {
656public:
663 DelayedBind(std::vector<std::shared_ptr<const Matrix<Value_, Index_> > > matrices, const bool by_row) :
664 my_matrices(std::move(matrices)), my_by_row(by_row)
665 {
666 auto nmats = my_matrices.size();
667 my_cumulative.reserve(sanisizer::sum<I<decltype(my_cumulative.size())> >(nmats, 1));
668 I<decltype(nmats)> sofar = 0;
669 my_cumulative.push_back(0);
670 Index_ min_extent = std::numeric_limits<Index_>::max();
671
672 for (I<decltype(nmats)> i = 0; i < nmats; ++i) {
673 auto& current = my_matrices[i];
674 Index_ primary, secondary;
675 if (my_by_row) {
676 primary = current->nrow();
677 secondary = current->ncol();
678 } else {
679 primary = current->ncol();
680 secondary = current->nrow();
681 }
682
683 if (i == 0) {
684 my_otherdim = secondary;
685 } else if (my_otherdim != secondary) {
686 throw std::runtime_error("all 'matrices' should have the same number of " + (my_by_row ? std::string("columns") : std::string("rows")));
687 }
688
689 // Removing the matrices that don't contribute anything,
690 // so we don't have to deal with their overhead.
691 if (primary > 0) {
692 if (sofar != i) {
693 my_matrices[sofar] = std::move(current);
694 }
695 my_cumulative.push_back(sanisizer::sum<Index_>(attest_for_Index(my_cumulative.back()), attest_for_Index(primary)));
696 min_extent = std::min(min_extent, primary);
697 ++sofar;
698 }
699 }
700
701 my_matrices.resize(sofar);
702 nmats = sofar;
703
704 if (nmats == 0) {
705 min_extent = 0;
706 }
707 my_mapping = DelayedBind_internal::CompressedMapping(min_extent, my_cumulative);
708
709 double denom = 0;
710 for (const auto& x : my_matrices) {
711 const double total = static_cast<double>(x->nrow()) * static_cast<double>(x->ncol());
712 denom += total;
713 my_sparse_prop += total * x->is_sparse_proportion();
714 my_by_row_prop += total * x->prefer_rows_proportion();
715 }
716 if (denom) {
717 my_sparse_prop /= denom;
718 my_by_row_prop /= denom;
719 }
720
721 for (int d = 0; d < 2; ++d) {
722 my_uses_oracle[d] = false;
723 for (const auto& x : my_matrices) {
724 if (x->uses_oracle(d)) {
725 my_uses_oracle[d] = true;
726 break;
727 }
728 }
729 }
730 }
